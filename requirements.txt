# Runtime dependencies for TripTokSummarizer.
# yt-dlp fetches caption tracks (and, optionally, audio).

# TikTok blocks plain HTTP clients, so yt-dlp must impersonate a real browser
# (`--impersonate chrome`). curl_cffi provides that and is required for the
# default captions source as well as the audio source.
curl_cffi>=0.5.10

# --- Summarization (optional, best quality) ---

# Requires the OPENAI_API_KEY environment variable.
openai>=1.0.0

# NOTE: The default `--source captions` needs no Whisper backend or ffmpeg.
